polyCoeffs

Coefficient matrices and column degrees of a polynomial matrix

Syntax

[Pk, c, d] = polyCoeffs(P, s)

Description

function [Pk, c, d] = polyCoeffs(P, s)
%POLYCOEFFS  Coefficient matrices and column degrees of a polynomial matrix.
%   [Pk, c, d] = polyCoeffs(P, s) returns the coefficient matrices of the
%   symbolic polynomial matrix P(s) = P_0 + P_1 s + ... + P_d s^d as the cell
%   array Pk, with Pk{k+1} = P_k (double), the column-degree vector c, where
%   c(j) is the degree of column j of P, and the degree d = max(c).
%
%   Example
%      syms s
%      P = [s^2 + 9*s + 11, 6; 7*s^2 + 13*s + 2, 4*s];
%      [Pk, c, d] = polyCoeffs(P, s)
%
%   See also HIGHESTCOLUMNCOEFF, GENSYLVESTER, TRIMMEDSYLVESTER.

P = expand(P);
[g, w] = size(P);
c = -ones(1, w);
for j = 1:w
    for r = 1:g
        if ~isequal(P(r, j), sym(0))
            c(j) = max(c(j), double(polynomialDegree(P(r, j), s)));
        end
    end
end
d = max(c);
Pk = repmat({zeros(g, w)}, 1, d + 1);
for r = 1:g
    for j = 1:w
        cf = fliplr(double(coeffs(P(r, j), s, 'All')));   % cf(k+1) = coefficient of s^k
        for k = 0:min(d, numel(cf) - 1)
            Pk{k+1}(r, j) = cf(k+1);
        end
    end
end
end

highestColumnCoeff

Highest column-degree coefficient matrix

Syntax

Phc = highestColumnCoeff(Pk, c)

Description

Phc = highestColumnCoeff(Pk, c) returns Phc, whose j-th column is the coefficient of scj in the j-th column of P(s). P(s) is column-reduced if and only if Phc has full column rank.

function Phc = highestColumnCoeff(Pk, c)
%HIGHESTCOLUMNCOEFF  Highest column-degree coefficient matrix P_hc.
%   Phc = highestColumnCoeff(Pk, c) returns the matrix whose j-th column is
%   the coefficient of s^c(j) in the j-th column of P(s). P(s) is
%   column-reduced if and only if rank(Phc) equals the number of columns.
%

[g, w] = size(Pk{1});
Phc = zeros(g, w);
for j = 1:w
    Phc(:, j) = Pk{c(j) + 1}(:, j);
end
end

genSylvester

Generalized Sylvester matrix Ai

Syntax

Description

function A = genSylvester(Pk, i)
%GENSYLVESTER  Generalized Sylvester matrix A_i of a polynomial matrix.
%   A = genSylvester(Pk, i) returns the generalized Sylvester matrix
%   A_i in R^{(i+1)g x (d+i+1)w} of P(s) = P_0 + P_1 s + ... + P_d s^d,
%   where Pk{k+1} = P_k. A_i is block Toeplitz with i+1 block rows
%
%      [ P_d  P_{d-1}  ...  P_0   0    ...  0  ]
%      [ 0    P_d      ...  P_1   P_0  ...  0  ]
%      [                  ...                  ]
%
%   i.e. A_i = A(1:(i+1)g, 1:(d+i+1)w) for the semi-infinite block Toeplitz
%   matrix A of [ELA, eq. (2.16)-(2.17)].
%
%   Example
%      [Pk, c, d] = polyCoeffs(P, s);
%      w = numel(c);
%      A = genSylvester(Pk, d*w - 1);     % the matrix of LAA Theorem 2.10
%
%   See also TRIMMEDSYLVESTER, POLYCOEFFS.

d = numel(Pk) - 1;
[g, w] = size(Pk{1});
A = zeros((i + 1)*g, (d + i + 1)*w);
for r = 0:i                         % block row r
    for k = 0:d                     % block column r+k holds P_{d-k}
        A(r*g + (1:g), (r + k)*w + (1:w)) = Pk{d - k + 1};
    end
end
end

trimmedSylvester

Trimmed Sylvester matrix Ti

Syntax

T = trimmedSylvester(Pk, c, i)[T, keep, removed] = trimmedSylvester(Pk, c, i)

Description

function [T, keep, removed] = trimmedSylvester(Pk, c, i)
%TRIMMEDSYLVESTER  Trimmed Sylvester matrix T_i of a polynomial matrix.
%   T = trimmedSylvester(Pk, c, i) returns the Trimmed Sylvester matrix
%   T_i in R^{(i+1)g x ((i+1)w + sum(c))} of P(s), where Pk{k+1} = P_k and
%   c is the column-degree vector. Column k*w + j of the generalized
%   Sylvester matrix A_i (block column k = 0, 1, ..., column j of P) contains
%   only coefficients of the j-th column of P(s) and is identically zero for
%   k < d - c(j). T_i is A_i with these d - c(j) columns removed for every
%   j [ELA, eq. (2.20), Definition 2.17].
%
%   [T, keep, removed] = trimmedSylvester(Pk, c, i) also returns the
%   indices of the columns of A_i that are kept (T = A_i(:, keep)) and
%   removed.
%
%   Example
%      [Pk, c] = polyCoeffs(P, s);
%      T = trimmedSylvester(Pk, c, sum(c) - 1);   % the matrix of ELA Theorem 3.8
%
%   See also GENSYLVESTER, POLYCOEFFS.

d = numel(Pk) - 1;
w = numel(c);
A = genSylvester(Pk, i);
keep = zeros(1, 0);
removed = zeros(1, 0);
for k = 0:d + i
    for j = 1:w
        q = k*w + j;
        if k >= d - c(j)
            keep(end + 1) = q;      %#ok<AGROW>
        else
            removed(end + 1) = q;   %#ok<AGROW>
        end
    end
end
T = A(:, keep);
end

function N = exactNull(M)
%EXACTNULL  Exact rational basis of the right nullspace of a matrix.
%   N = exactNull(M) returns a symbolic matrix whose columns form a basis of
%   {x : M*x = 0}, computed in exact rational arithmetic from the reduced
%   row echelon form R = rref(sym(M)): for every free (non-pivot) column f
%   there is one basis vector with 1 in position f, 0 in the other free
%   positions, and -R(q, f) in the position of the q-th pivot.
%   size(N, 2) is the nullity of M.
%
%   Example
%      T = trimmedSylvester(Pk, c, sum(c) - 1);
%      N = exactNull(T);
%      nullity = size(N, 2)
%      isequal(sym(T)*N, sym(zeros(size(T, 1), nullity)))   % check
%
%   See also NULL, RREF, GENSYLVESTER, TRIMMEDSYLVESTER.

S = sym(M);
n = size(S, 2);
R = rref(S);
nz = double(R) ~= 0;
piv = zeros(1, 0);
for r = 1:size(R, 1)
    p = find(nz(r, :), 1);
    if isempty(p), break; end
    piv(end + 1) = p; %#ok<AGROW>
end
free = setdiff(1:n, piv);
N = sym(zeros(n, numel(free)));
for t = 1:numel(free)
    f = free(t);
    N(f, t) = 1;
    for q = 1:numel(piv)
        N(piv(q), t) = -R(q, f);
    end
end
end

function [idx, L, S] = leftMinimalIndices(Pk, c)
%LEFTMINIMALINDICES  Left minimal indices from the Trimmed Sylvester matrices.
%   [idx, L, S] = leftMinimalIndices(Pk, c) computes the left nullities
%   eta_i = (#rows of T_i) - rank(T_i) of the Trimmed Sylvester matrices
%   T_0, T_1, ..., T_{sum(c)}, their first differences
%   eta1_i = eta_i - eta_{i-1} and second differences
%   eta2_i = eta1_i - eta1_{i-1}. By [ELA, Theorem 3.1], eta2_i of the
%   left minimal indices of P(s) are equal to i. The function returns
%      idx  the left minimal indices (a row vector of g - w integers),
%      L    their sum L_P,
%      S    a table with columns i, size, rank, eta, eta1, eta2 and the left
%           rank deficiency kappa = min(size(T_i)) - rank(T_i).
%   Ranks are exact (Symbolic Math Toolbox).
%
%   Example
%      [Pk, c] = polyCoeffs(P, s);
%      [idx, L, S] = leftMinimalIndices(Pk, c)
%
%   See also TRIMMEDSYLVESTER.

n = sum(c) + 1;
i = (0:n-1)';
rows = zeros(n, 1); cols = zeros(n, 1); rk = zeros(n, 1);
for t = 1:n
    T = trimmedSylvester(Pk, c, t - 1);
    [rows(t), cols(t)] = size(T);
    rk(t) = rank(sym(T));
end
eta   = rows - rk;
eta1  = [eta(1);  diff(eta)];
eta2  = [eta1(1); diff(eta1)];
kappa = min(rows, cols) - rk;
idx = repelem(i', eta2');
L = sum(idx);
sz = strcat(string(rows), "x", string(cols));
S = table(i, sz, rk, eta, eta1, eta2, kappa, ...
    'VariableNames', {'i', 'size', 'rank', 'eta', 'eta1', 'eta2', 'kappa'});

function [detG, MP, minors, rows] = maxMinorGcd(P, s)
%MAXMINORGCD  Maximal minors of a tall polynomial matrix and their gcd.
%   [detG, MP] = maxMinorGcd(P, s) returns the monic gcd detG of all w x w
%   (maximal) minors of the g x w polynomial matrix P(s), and the
%   maximal-minor degree MP. detG is the product of the invariant factors of
%   P(s); it equals det G(s), up to a nonzero constant, for every GCRD G(s)
%   of the blocks of P(s). Hence sigma_f(P) = polynomialDegree(detG).
%
%   [detG, MP, minors, rows] = maxMinorGcd(P, s) also returns all maximal
%   minors and, in row k of rows, the rows of P used for minors(k).
%
%   Example
%      [detG, MP] = maxMinorGcd(P, s);
%      sigmaF = polynomialDegree(detG)
%      factor(detG)
%
%   See also INFEIGENVALUES, GCD, DET.

[g, w] = size(P);
rows = nchoosek(1:g, w);
minors = sym(zeros(1, size(rows, 1)));
deg = -ones(1, size(rows, 1));
for k = 1:size(rows, 1)
    minors(k) = expand(det(P(rows(k, :), :)));
    if ~isequal(minors(k), sym(0))
        deg(k) = double(polynomialDegree(minors(k), s));
    end
end
MP = max(deg);
detG = minors(1);
for k = 2:numel(minors)
    detG = gcd(detG, minors(k));
end
detG = expand(detG);
if ~isequal(detG, sym(0))
    cf = coeffs(detG, s, 'All');
    detG = expand(detG / cf(1));             % monic
end
end

function [sigInfC, sigInf, revcP, revP] = infEigenvalues(Pk, c, s)
%INFEIGENVALUES  Number of infinite eigenvalues w.r.t. c and w.r.t. d.
%   [sigInfC, sigInf] = infEigenvalues(Pk, c, s) returns
%      sigInfC  sigma_inf^(c)(P): the multiplicity of the eigenvalue 0 of the
%               column-degree reversal rev_c P(s) = P(1/s) diag(s^c(1), ..., s^c(w)),
%      sigInf   sigma_inf(P): the same for the reversal rev P(s) = s^d P(1/s).
%   Each is the multiplicity of the root s = 0 of the gcd of the maximal
%   minors of the reversal. The two satisfy
%      sigInf = sigInfC + sum(d - c)        [ELA, eq. (2.12)],
%   and sigInfC = 0 if and only if P(s) is column-reduced.
%
%   [sigInfC, sigInf, revcP, revP] = infEigenvalues(Pk, c, s) also returns
%   the two reversals.
%
%   Example
%      [Pk, c] = polyCoeffs(P, s);
%      [sigInfC, sigInf] = infEigenvalues(Pk, c, s)
%
%   See also MAXMINORGCD, POLYCOEFFS.

d = numel(Pk) - 1;
[g, w] = size(Pk{1});
revcP = sym(zeros(g, w));
revP  = sym(zeros(g, w));
for r = 1:g
    for j = 1:w
        for l = 0:c(j), revcP(r, j) = revcP(r, j) + Pk{l+1}(r, j) * s^(c(j) - l); end
        for l = 0:d,    revP(r, j)  = revP(r, j)  + Pk{l+1}(r, j) * s^(d - l);    end
    end
end
sigInfC = orderAtZero(maxMinorGcd(revcP, s), s);
sigInf  = orderAtZero(maxMinorGcd(revP, s), s);
end

function o = orderAtZero(f, s)
% multiplicity of s = 0 as a root of the nonzero polynomial f
cf = double(coeffs(expand(f), s, 'All'));     % highest power first
o = numel(cf) - find(cf ~= 0, 1, 'last');
end
